void drawAlarmIcon16(int x, int y, uint16_t color) {
  int cx = x + 8;
  int cy = y + 8;

  // tělo budíku
  tft.fillCircle(cx, cy, 7, color);
  tft.drawCircle(cx, cy, 7, TFT_BLACK);

  // ručičky
  tft.drawLine(cx, cy, cx, cy - 4, TFT_BLACK);
  tft.drawLine(cx, cy, cx + 3, cy, TFT_BLACK);
  tft.fillCircle(cx, cy, 1, TFT_BLACK);

  // zvonky
  tft.fillCircle(cx - 5, y + 2, 2, color);
  tft.fillCircle(cx + 5, y + 2, 2, color);
}

uint16_t getIndoorTempColor(float t) {
  if (isnan(t)) return TFT_WHITE;
  if (t < 18.0) return TFT_BLUE;			// teplotní prahy + barva písma
  if (t < 22.0) return TFT_CYAN;
  if (t < 24.0) return TFT_GREEN;
  if (t <= 26.0) return TFT_YELLOW;
  return TFT_RED;
}

}

unsigned long lastButtonActionMillis = 0;
const unsigned long BUTTON_DEBOUNCE_TIME = 250;  // ms

bool buttonActionAllowed() {
  unsigned long now = millis();

  if (now < buttonsBlockedUntil) return false;  // blokace po alarmu
  if (now - lastButtonActionMillis < BUTTON_DEBOUNCE_TIME) return false;

  lastButtonActionMillis = now;
  return true;
}

// ===== DEBOUNCE pro UP/DOWN =====
bool buttonPressed(int pin) {
  static unsigned long lastPress[40];
  if (digitalRead(pin) == LOW) {
    if (millis() - lastPress[pin] > 300) {
      lastPress[pin] = millis();
      return true;
    }
  }
  return false;
}

  prefs.putUChar("hour", alarm1.hour);
  prefs.putUChar("minute", alarm1.minute);
  prefs.putBool("enabled", alarm1.enabled);
  prefs.end();
}

void handleAlarmControl() {
  // zapnutí / vypnutí
  if (server.hasArg("enable")) {
    bool enable = server.arg("enable") == "1";
    alarm1.enabled = enable;

    // 🔴 pokud web vypíná alarm → kompletní reset
    if (!enable) {
      resetAfterAlarm();
    }

    saveAlarm();
    redraw = true;
  }

  // nastavení času HH:MM
  if (server.hasArg("time")) {
    String t = server.arg("time");
    if (t.length() == 5 && t.charAt(2) == ':') {
      alarm1.hour = t.substring(0, 2).toInt();
      alarm1.minute = t.substring(3, 5).toInt();
    saveAlarm();
    redraw = true;
    }
  }

server.sendHeader("Connection", "close");
  server.send(200, "text/plain", "OK");
}

// ===== OVLÁDÁNÍ TLAČÍTEK =====
bool handleButtons(bool& alarmActive, bool nightNow) {

  // 🔴 ochrana proti rychlému mačkání
  if (millis() < buttonsBlockedUntil) return false;

  // ===== BLOKACE TLAČÍTEK PO VYPNUTÍ ALARMU =====
  if (millis() < buttonsBlockedUntil) {
    return false;
  }

  static unsigned long setPressTime = 0;
  static bool setHeld = false;
  static bool longActionDone = false;
  bool redrawNeeded = false;

  // Pokud budík běží, tlačítka pouze deaktivují alarm
  // ===== BLOKACE UI během celého buzení =====
  if (alarmRunning) {

    static unsigned long pressStart = 0;
    static bool pressed = false;

    // detekce stisku SET
    if (digitalRead(BTN_SET) == LOW) {
      if (!pressed) {
        pressed = true;
        pressStart = millis();
      }

      // ===== DLOUHÝ STISK → úplné vypnutí =====
      if (pressed && millis() - pressStart > 2000) {

        alarmActive = false;

        snoozeCount = 0;
        alarm1.enabled = false;
        saveAlarm();

        resetAfterAlarm();

        pressed = false;
        return true;
      }
    } else {  // 🔴 TOTO patří k if(digitalRead...)

      // ===== KRÁTKÝ STISK =====
      if (pressed) {

        pressed = false;

        if (snoozeTriggeredThisCycle) return false;
        snoozeTriggeredThisCycle = true;

        alarmActive = false;
        wakeProcessActive = false;
        dfPlayer.stop();
        dfPlayer.volume(0);

        snoozeCount++;

        if (snoozeCount < 3) {
          snoozeActive = true;

          if (snoozeCount == 1)
            nextAlarmMillis = millis() + SNOOZE_1;
          else
            nextAlarmMillis = millis() + SNOOZE_2;
        } else {
          snoozeActive = false;
          snoozeCount = 0;
          wakeProcessActive = false;

          alarm1.enabled = false;
          saveAlarm();
        }

        buttonsBlockedUntil = millis() + BUTTON_BLOCK_TIME;
        return true;
      }
    }

    // ===== UP / DOWN → taky snooze =====
    if ((buttonPressed(BTN_UP) || buttonPressed(BTN_DOWN))) {

      if (snoozeTriggeredThisCycle) return false;
      snoozeTriggeredThisCycle = true;

      alarmActive = false;
      wakeProcessActive = false;
      dfPlayer.stop();
      dfPlayer.volume(0);

      snoozeCount++;

      if (snoozeCount < 3) {
        snoozeActive = true;

        if (snoozeCount == 1)
          nextAlarmMillis = millis() + SNOOZE_1;
        else
          nextAlarmMillis = millis() + SNOOZE_2;
      } else {
        snoozeActive = false;
        snoozeCount = 0;
        wakeProcessActive = false;

        alarm1.enabled = false;
        saveAlarm();
      }

      buttonsBlockedUntil = millis() + BUTTON_BLOCK_TIME;
      return true;
    }

    return false;
  }

  // ===== NOČNÍ REŽIM – speciální chování =====
  if (nightNow && !nightModeOverride) {

    // jakékoliv tlačítko = rozsvítit displej
    if (digitalRead(BTN_SET) == LOW || digitalRead(BTN_UP) == LOW || digitalRead(BTN_DOWN) == LOW) {

      nightWakeActive = true;
      nightWakeUntil = millis() + NIGHT_WAKE_TIME;
    }

   // ===== SET – detekce držení =====
    if (digitalRead(BTN_SET) == LOW && !setHeld) {
      setHeld = true;
      setPressTime = millis();
      longActionDone = false; //  Na začátku stisku vyresetujeme pojistku
    }

    // dlouhý stisk >2s → vypnutí nočního režimu
    if (setHeld && digitalRead(BTN_SET) == LOW && millis() - setPressTime > 2000 && !longActionDone) {

      nightModeOverride = true;
      nightWakeActive = false;

      brightness = normalBrightness;
      ledcWrite(BACKLIGHT_PIN, brightness);

      longActionDone = true; //  Říkáme systému, ať ignoruje puštění tlačítka
      buttonsBlockedUntil = millis() + 500;
      return true; //  Změnili jsme jas, chceme obrazovku hned překreslit
    }

    // uvolnění
    if (digitalRead(BTN_SET) == HIGH) {
      setHeld = false;
    }

    // ❌ v nočním režimu NIC jiného nedělej
    return false;
  }

  int btnState = digitalRead(BTN_SET);

  // tlačítko právě stisknuto
  if (btnState == LOW && !setHeld) {
    setPressTime = millis();
    setHeld = true;
    longActionDone = false;
  }

  // tlačítko drženo >3s -> dlouhý stisk (přepnutí budíku)
  if (setHeld && digitalRead(BTN_SET) == LOW && millis() - setPressTime >= 3000 && !longActionDone && buttonActionAllowed()) {

    alarm1.enabled = !alarm1.enabled;

    //==== DEFINITIVNÍ VYPNUTÍ =====
    alarmActive = false;
    snoozeActive = false;
    snoozeCount = 0;
    nextAlarmMillis = 0;

    wakeProcessActive = false;  //  ukončení buzení

    dfPlayer.stop();
    dfPlayer.volume(0);

    if (alarmBrightnessForced) {
      brightness = normalBrightness;
      ledcWrite(BACKLIGHT_PIN, brightness);
      alarmBrightnessForced = false;
    }

    saveAlarm();
    longActionDone = true;
    redrawNeeded = true;
  }

  // tlačítko uvolněno
  if (btnState == HIGH && setHeld) {
    unsigned long pressDuration = millis() - setPressTime;
    setHeld = false;
    setPressTime = 0;

    // ❗ pokud už proběhl long press → NIC nedělej
    if (longActionDone) {
      longActionDone = false;
      return redrawNeeded;
    }

    if (!buttonActionAllowed()) return redrawNeeded;

    if (pressDuration >= 50 && pressDuration < 1000) {
	lastSettingActivityMillis = millis();
      if (uiState == SHOW_TIME) {
        uiState = SET_HOUR;
        userIsSettingAlarm = true;   // 🚩 PRÁVĚ JSI VSTOUPIL DO NASTAVENÍ
      } 
      else if (uiState == SET_HOUR) {
        uiState = SET_MINUTE;
        // userIsSettingAlarm zůstává true
      } 
      else {
        uiState = SHOW_TIME;
        userIsSettingAlarm = false;  // 🚩 PRÁVĚ JSI VYŠEL Z NASTAVENÍ (Budík může začít hlídat)
        saveAlarm();
        lastAlarmMinute = -1;
      }
      redrawNeeded = true;
    }

  }

  // UP tlačítko
  if (buttonPressed(BTN_UP) && buttonActionAllowed()) {
	lastSettingActivityMillis = millis();
    // 🔆 REGULACE JASU – pouze při normálním zobrazení
    if (uiState == SHOW_TIME) {
      uint16_t step = getBrightnessStep(brightness);

      if (brightness + step <= PWM_MAX)
        brightness += step;
      else
        brightness = PWM_MAX;

      ledcWrite(BACKLIGHT_PIN, brightness);
      saveBrightness();
    }

    // ⏰ NASTAVENÍ BUDÍKU (zachováno!)
    if (uiState == SET_HOUR) alarm1.hour = (alarm1.hour + 1) % 24;
    if (uiState == SET_MINUTE) alarm1.minute = (alarm1.minute + 1) % 60;

    redrawNeeded = true;
  }

  // DOWN tlačítko
  if (buttonPressed(BTN_DOWN) && buttonActionAllowed()) {
	lastSettingActivityMillis = millis();
    // 🔆 REGULACE JASU – pouze při normálním zobrazení
    // DOWN
    if (uiState == SHOW_TIME) {
      uint16_t step = getBrightnessStep(brightness);

      if (brightness >= step)
        brightness -= step;
      else
        brightness = 0;

      ledcWrite(BACKLIGHT_PIN, brightness);
      saveBrightness();
    }

    // ⏰ NASTAVENÍ BUDÍKU (zachováno!)
    if (uiState == SET_HOUR) alarm1.hour = (alarm1.hour + 23) % 24;
    if (uiState == SET_MINUTE) alarm1.minute = (alarm1.minute + 59) % 60;

    redrawNeeded = true;
  }

  return redrawNeeded;
}

// ==== po buzeni návrat do normálu ====

void resetAfterAlarm() {

  alarmRunning = false;
  wakeProcessActive = false;
  snoozeActive = false;
  snoozeCount = 0;
  nightModeOverride = true;

  dfPlayer.stop();
  dfPlayer.volume(0);

  // 🔴 DŮLEŽITÉ – vypnout forced brightness
  alarmBrightnessForced = false;

  // 🔆 návrat jasu
  brightness = normalBrightness;
  ledcWrite(BACKLIGHT_PIN, brightness);

  // 🔘 UI zpět
  uiState = SHOW_TIME;
  userIsSettingAlarm = false;  // Jistota, že budík začne hlídat čas
  lastAlarmMinute = -1;       //  Reset paměti poslední minuty buzení

  // 🔴 reset cyklu tlačítek
  snoozeTriggeredThisCycle = false;

  // ochrana proti ghost klikům
  buttonsBlockedUntil = millis() + BUTTON_BLOCK_TIME;
}

void updateBrightnessForNightMode(struct tm& timeinfo) {

  // 🔴 ALARM má nejvyšší prioritu
  if (alarmBrightnessForced) {
    ledcWrite(BACKLIGHT_PIN, PWM_MAX);
    return;
  }

  bool active = false;

  int current = timeinfo.tm_hour * 60 + timeinfo.tm_min;
  int start = nightStartHour * 60 + nightStartMinute;
  int end = nightEndHour * 60 + nightEndMinute;

  if (start < end) active = current >= start && current < end;
  else active = current >= start || current < end;

  // 🔁 reset override mimo noční režim
  if (!active) {
    nightModeOverride = false;
  }

  // 👉 override jen potlačí aktivaci, ale NEUKONČÍ funkci
  if (nightModeOverride) {
    nightModeActive = false;
    brightness = normalBrightness;
  } else {
    nightModeActive = active;
    brightness = active ? nightBrightness : normalBrightness;
  }

  // wake má prioritu
  if (nightWakeActive) {
    if (millis() < nightWakeUntil) {
      ledcWrite(BACKLIGHT_PIN, PWM_MAX);
      return;
    } else {
      nightWakeActive = false;
    }
  }

  ledcWrite(BACKLIGHT_PIN, brightness);

  // 🔁 reset override přes den
  if (!active) {
    nightModeOverride = false;
  }
}

void drawAlarmBlink(bool alarmActive) {

  if (currentMillis - lastBlinkTime > 500) {
    blinkState = !blinkState;
    lastBlinkTime = currentMillis;
  }

  tft.setTextColor(TFT_WHITE);

  sprintf(hourStr, "%02d", alarm1.hour);
  sprintf(minStr, "%02d", alarm1.minute);

  if (uiState == SET_HOUR) {
    tft.setTextColor(blinkState ? TFT_GREEN : TFT_BLACK);
    tft.drawString(hourStr, 245, 215);  // nastavení kde bude čas budíku + další pole dolu
    tft.setTextColor(TFT_GREEN);

    tft.drawString(":", 275, 215);
    tft.setTextColor(blinkState ? TFT_GREEN : TFT_BLACK);

  } else {
    tft.setTextColor(alarm1.enabled ? TFT_GREEN : TFT_RED);

    tft.drawString(alarmStr, 245, 215);  // konec nastavení času budíku

  if (!alarm1.enabled) return;

  uint16_t alarmColor;

  if (alarmRunning) {
    alarmColor = TFT_RED;
  } else {
    alarmColor = TFT_GREEN;
  }

  drawAlarmIcon16(160, 15, alarmColor);
}

void fetchTodayName() {
  if (WiFi.status() != WL_CONNECTED) {
    Serial.println("WiFi neni pripojena");
    return;
  }

  WiFiClientSecure client;
  client.setInsecure();  // ignoruje validitu certifikátu

  HTTPClient https;
  https.begin(client, "https://svatkyapi.cz/api/day");  // HTTPS
  https.setTimeout(2000); // Nastaví maximální dobu čekání na 2 sekundy (2000 ms)
  int httpCode = https.GET();
  Serial.print("HTTP kod: ");
  Serial.println(httpCode);

  if (httpCode == 200) {
    String payload = https.getString();
    Serial.println("DATA ZE SERVERU:");
    Serial.println(payload);

    DynamicJsonDocument doc(512);
    DeserializationError error = deserializeJson(doc, payload);
    if (error) {
      Serial.println("JSON chyba");
      todayName = "";
    } else {
      // použít správný klíč
      todayName = removeDiacritics(doc["name"].as<String>());
      Serial.print("Svatek: ");
      Serial.println(todayName);
    }
  } else {
    Serial.println("Chyba pri HTTP requestu");
    todayName = "";
  }

  https.end();
}

bool isNightNow(struct tm& timeinfo) {
  int current = timeinfo.tm_hour * 60 + timeinfo.tm_min;
  int start = nightStartHour * 60 + nightStartMinute;
  int end = nightEndHour * 60 + nightEndMinute;

  if (start < end){
    return current >= start && current < end;
  } else {
    return current >= start || current < end;
  }
}

  pinMode(BTN_SET, INPUT_PULLUP);
  pinMode(BTN_UP, INPUT_PULLUP);
  pinMode(BTN_DOWN, INPUT_PULLUP);
  pinMode(POWER_DETECT_PIN, INPUT_PULLDOWN);
  pinMode(ALARM_PIN_LEFT, OUTPUT);
  
  // PWM pro noční světýlko (frekvence 1 kHz, rozlišení 8-bit tedy 0-255)
  ledcAttach(NIGHT_LIGHT_PIN, 1000, 8); 
  ledcWrite(NIGHT_LIGHT_PIN, 0); // Po startu raději vypnout
  
  loadNightLight(); // Načtení dat z paměti

  server.enableDelay(false);
  
  tft.init();
  tft.setRotation(1);
  tft.fillScreen(TFT_BLACK);
  tft.setTextSize(2);
  tft.setTextColor(TFT_GREEN);
  tft.drawString("Spoustim...", 20, 20);

  loadAlarm();

  // ===== PWM PODSVIT (ESP32 core 3.x) =====
  ledcAttach(BACKLIGHT_PIN, PWM_FREQ, PWM_RESOLUTION);
  ledcWrite(BACKLIGHT_PIN, brightness);

  loadBrightness();
  loadNightMode();  // načtení nočního režimu z Preferences
  ledcWrite(BACKLIGHT_PIN, brightness);

  // ===== WiFi s pevnou IP =====
  tft.drawString("WiFi...", 20, 50);
  WiFi.mode(WIFI_STA);
  delay(100);

  IPAddress local_IP(192, 168, 0, 232);  // pevná IP,jakou zjistím v cmd-ipconfig 	//
  IPAddress gateway(192, 168, 0, 1);     // adresa routeru				//
  IPAddress subnet(255, 255, 255, 0);    // maska sítě					// nastavit IP	
  IPAddress primaryDNS(8, 8, 8, 8);      // Google DNS					//
  IPAddress secondaryDNS(8, 8, 4, 4);    // Google DNS					//

  if (!WiFi.config(local_IP, gateway, subnet, primaryDNS, secondaryDNS)) {
    Serial.println("Chyba nastavení statické IP");
  }

  // připojení k Wi-Fi
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) delay(500);
  tft.drawString("WiFi OK", 20, 80);

  // načtení svátku hned o startu
  fetchTodayName();
  lastNameFetch = millis();

  // spouštění web serveru
  server.on("/", handleRoot);
  server.on("/data", handleData);
  server.on("/favicon.ico", []() { server.send(200, "image/svg+xml", R"rawliteral(<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 100 100"><text y="0.9em" font-size="90">⏰</text></svg>)rawliteral"); });
  server.on("/update", HTTP_GET, handleUpdate);
  server.begin();

  // vypíše IP do Serial monitoru
  Serial.print("HTTP server na IP: ");
  Serial.println(WiFi.localIP());

  // ==== ovládání budíku z webu ====
  server.on("/alarm", HTTP_GET, handleAlarmControl);

  // nastavení jasu z webu
  server.on("/getNightMode", HTTP_GET, []() {
    String json = "{";
    json += "\"start\":\"" + twoDigits(nightStartHour) + ":" + twoDigits(nightStartMinute) + "\",";
    json += "\"end\":\"" + twoDigits(nightEndHour) + ":" + twoDigits(nightEndMinute) + "\",";
    json += "\"brightness\":" + String(nightBrightness);
    json += "}";
    server.sendHeader("Connection", "close");
    server.send(200, "application/json", json);
  });

  server.on("/brightness", HTTP_GET, []() {
    if (server.hasArg("value")) {
      int perc = server.arg("value").toInt();
      if (perc < 0) perc = 0;
      if (perc > 100) perc = 100;

      normalBrightness = (uint16_t)((perc * PWM_MAX) / 100);
      brightness = normalBrightness;
      ledcWrite(BACKLIGHT_PIN, brightness);
      saveBrightness();
    }
    server.sendHeader("Connection", "close");
    server.send(200, "text/plain", "OK");
  });

  // nastavení nočního režimu
  server.on("/nightmode", HTTP_GET, []() {
    if (server.hasArg("start") && server.hasArg("end") && server.hasArg("brightness")) {
      String s = server.arg("start");  // HH:MM
      String e = server.arg("end");
      int b = server.arg("brightness").toInt();

      nightStartHour = s.substring(0, 2).toInt();
      nightStartMinute = s.substring(3, 5).toInt();
      nightEndHour = e.substring(0, 2).toInt();
      nightEndMinute = e.substring(3, 5).toInt();
      if (b < 0) b = 0;
      if (b > 4095) b = 4095;
      nightBrightness = b;

      saveNightMode();
    }
    server.sendHeader("Connection", "close");
    server.send(200, "text/plain", "OK");
  });

// Načtení dat o světýlku do webu
  server.on("/getNL", HTTP_GET, []() {
    String json = "{";
    json += "\"start\":\"" + twoDigits(nlStartHour) + ":" + twoDigits(nlStartMinute) + "\",";
    json += "\"end\":\"" + twoDigits(nlEndHour) + ":" + twoDigits(nlEndMinute) + "\",";
    json += "\"brightness\":" + String(nlBrightness);
    json += "}";
    server.sendHeader("Connection", "close");
    server.send(200, "application/json", json);
  });

  // Uložení dat o světýlku z webu
  server.on("/setNL", HTTP_GET, []() {
    if (server.hasArg("start") && server.hasArg("end") && server.hasArg("brightness")) {
      String s = server.arg("start");
      String e = server.arg("end");
      nlStartHour = s.substring(0, 2).toInt();
      nlStartMinute = s.substring(3, 5).toInt();
      nlEndHour = e.substring(0, 2).toInt();
      nlEndMinute = e.substring(3, 5).toInt();
      
      int b = server.arg("brightness").toInt();
      if (b < 0) b = 0;
      if (b > 100) b = 100;
      nlBrightness = b;

      saveNightLight();
    }
    server.sendHeader("Connection", "close");
    server.send(200, "text/plain", "OK");
  });  

  // NTP
  configTzTime("CET-1CEST,M3.5.0/2,M10.5.0/3", ntpServer);
  tft.drawString("Cas...", 20, 110);
  struct tm timeinfo;
  int retry = 0;
  while (!getLocalTime(&timeinfo) && retry < 20) {
    delay(500);
    retry++;
  }
  if (retry < 20) tft.drawString("Cas OK", 20, 140);
  else tft.drawString("Cas FAIL", 20, 140);

  delay(1000);

  WiFi.mode(WIFI_STA);
  WiFi.setSleep(false);

  tft.fillScreen(TFT_BLACK);

  // ===== DFPlayer =====
  dfSerial.begin(9600, SERIAL_8N1, 16, 17);  // RX=16, TX=17
  if (!dfPlayer.begin(dfSerial)) {
    Serial.println("DFPlayer nenalezen!");
  } else {
    dfPlayer.volume(5);  // hlasitost 0-30          // hlasitost , bylo nastaveno 20
  }

// ===== DS18B20 INIT =====
  sensorsInternal.begin();

  // ===== WATCHDOG (ZAPNE SE AŽ PO ÚSPĚŠNÉM STARTU) =====
  esp_task_wdt_config_t wdt_config = { .timeout_ms = 15000, .idle_core_mask = (1 << portNUM_PROCESSORS) - 1, .trigger_panic = true };
  esp_task_wdt_init(&wdt_config); 
  esp_task_wdt_add(NULL); 
}

//============================================================================================
void loop() {
  esp_task_wdt_reset(); 
  
  // 1. Vše kolem aktivní WiFi (web server, reconnect i kontrolu IP) řešíme JEN, pokud má být WiFi zapnutá
  if (wifiEnabled) {
    server.handleClient();

    // A. Bezpečná kontrola a obnova WiFi (maximálně jednou za 10 vteřin)
    // Používáme tiché reconnect(), které neshazuje celou kartu a nezatěžuje procesor
    static unsigned long lastWiFiCheck = 0;
    if (WiFi.status() != WL_CONNECTED) {
      if (millis() - lastWiFiCheck > 10000UL) { 
        lastWiFiCheck = millis();
        WiFi.reconnect(); 
        Serial.println("WiFi odpojena, zkousim tichy reconnect...");
      }
    }

    // B. KONTROLA SPRÁVNÉ IP ADRESY (každých 5 minut)
    // Spouští se jen když jsme připojení. Hlídá, zda router nezměnil naši pevnou IP.
    static unsigned long lastIpCheck = 0;
    if (WiFi.status() == WL_CONNECTED && (millis() - lastIpCheck > 300000UL)) {
      lastIpCheck = millis();
      if (WiFi.localIP() != local_IP) {
        Serial.println("Chybna IP adresa! Restartuji ESP...");
        delay(500);    // Krátká pauza, aby se stihl odeslat text do Serial linky
        ESP.restart(); // Tvrdý restart jako záchranná brzda
      }
    }
  }

  // ===== DETEKCE NAPÁJENÍ (WiFi On/Off) =====
  bool powerPresent = digitalRead(POWER_DETECT_PIN);
  
  // Došlo k odpojení ze sítě -> vypínáme WiFi kvůli úspoře baterie
  if (!powerPresent && wifiEnabled) {
    Serial.println("Vypadek napajeni - vypinam WiFi");
    server.close();
    WiFi.disconnect(true);
    WiFi.mode(WIFI_OFF);
    wifiEnabled = false;
  }
  
  // Napájení se vrátilo -> zapínáme WiFi zpět
  if (powerPresent && !wifiEnabled) {
    Serial.println("Napajeni OK - zapinam WiFi");
    WiFi.mode(WIFI_STA);
    WiFi.config(local_IP, gateway, subnet, primaryDNS, secondaryDNS);
    WiFi.begin(ssid, password);
    
    unsigned long start = millis();
    while (WiFi.status() != WL_CONNECTED && millis() - start < 5000) {
      // server.handleClient(); // OPRAVENO: Odstraněno, server ještě neběží, způsobovalo to pády ESP32!
      delay(100);
    }
    if (WiFi.status() == WL_CONNECTED) {
      server.begin();
      Serial.println("WiFi uspesne obnovena a server bezi.");
    }
    wifiEnabled = true;
  }

// ===== STAV ALARMU A ČAS =====
  static bool firstDraw = true;
  static unsigned long alarmStartMillis = 0;
  alarmRunning = wakeProcessActive || snoozeActive;
  
  struct tm timeinfo;
  bool timeValid = getLocalTime(&timeinfo);

  static bool lastAlarmRunning = false;
  if (alarmRunning != lastAlarmRunning) {
    redraw = true;
    lastAlarmRunning = alarmRunning;
  }

  // ==== AUTOMATICKÝ RESET NOČNÍHO REŽIMU POD ČASU NASTAVENEHO NA WEBU ====
  if (timeValid && timeinfo.tm_hour == 22 && timeinfo.tm_min == 0 && timeinfo.tm_sec < 5) {
    if (nightModeOverride) {
      nightModeOverride = false;
      redraw = true; // Překreslíme, aby se projevila změna jasu
    }
  }

  // ==== DFPlayer úplně ticho mimo alarm ====
  // OPRAVENO: Hlídáme si změnu stavu (lastAlarmActiveState), příkaz odešleme pouze JEDNOU při vypnutí budíku!
  static bool lastAlarmActiveState = false;
  if (!alarmActive && lastAlarmActiveState) {
    dfPlayer.stop();
    dfPlayer.volume(0);
  }
  lastAlarmActiveState = alarmActive;

// ===== OČI – LED diody (Budík) =====
  digitalWrite(ALARM_PIN_LEFT, alarmRunning ? HIGH : LOW);

  // ===== NOČNÍ SVĚTÝLKO =====
  if (timeValid) {
    int currentMin = timeinfo.tm_hour * 60 + timeinfo.tm_min;
    int startMin = nlStartHour * 60 + nlStartMinute;
    int endMin = nlEndHour * 60 + nlEndMinute;

    bool nlActiveNow = false;
    // Logika zohledňující, zda interval přechází přes půlnoc
    if (startMin < endMin) {
      nlActiveNow = (currentMin >= startMin && currentMin < endMin);
    } else {
      nlActiveNow = (currentMin >= startMin || currentMin < endMin);
    }

    if (nlActiveNow) {
      // Přepočet 0-100% na hodnotu PWM 0-255 pro 8-bit rozlišení
      int pwmValue = map(nlBrightness, 0, 100, 0, 255);
      ledcWrite(NIGHT_LIGHT_PIN, pwmValue);
    } else {
      ledcWrite(NIGHT_LIGHT_PIN, 0); // Mimo čas je světýlko zhasnuté
    }
  }

  // Automatické přepínání jasu a NTP synchronizace
  if (timeValid) {
    updateBrightnessForNightMode(timeinfo);
    if (millis() - lastNtpSyncMillis > NTP_SYNC_INTERVAL) {
      lastNtpSyncMillis = millis();
      syncTimeWithNTP();
    }
  }

  // Měření teploty (každých 5s)
  static unsigned long lastTempRead = 0;
  if (millis() - lastTempRead > 5000) {
    lastTempRead = millis();
    sensorsInternal.requestTemperatures();
    
    float t = sensorsInternal.getTempCByIndex(0);
    if (t > -50 && t < 85) {
      tempInternal = t + INTERNAL_TEMP_OFFSET; 
    } else {
      tempInternal = NAN;
    }
  }

  bool stopAlarm = false;

  // Svátek/Jméno dne (každou hodinu)
  if (millis() - lastNameFetch > NAME_FETCH_INTERVAL) {
    lastNameFetch = millis();
    fetchTodayName();
  }

  // ===== SPUŠTĚNÍ ALARMU =====
  if (timeValid && alarm1.enabled && !alarmActive && !snoozeActive && !userIsSettingAlarm) {
    int currentMinutes = timeinfo.tm_hour * 60 + timeinfo.tm_min;
    int alarmMinutes = alarm1.hour * 60 + alarm1.minute;

    if (currentMinutes == alarmMinutes && lastAlarmMinute != alarmMinutes) {
      snoozeTriggeredThisCycle = false;
      alarmActive = true;
      wakeProcessActive = true;
      alarmRunning = true;     
      lastAlarmMinute = alarmMinutes;
      alarmStartMillis = millis();
      if (!alarmBrightnessForced) {
        ledcWrite(BACKLIGHT_PIN, PWM_MAX);
        alarmBrightnessForced = true;
      }

      dfPlayer.volume(VOLUME_START);
      int nahodnaPisen = random(1, 6);  // Vybere náhodné číslo od 1 do 5.
      dfPlayer.loop(nahodnaPisen);
      redraw = true;    
    }
  }

  // ===== SNOOZE LOGIKA =====
  if (snoozeActive && !alarmActive && millis() >= nextAlarmMillis) {
    snoozeTriggeredThisCycle = false;
    alarmActive = true;
    wakeProcessActive = true;
    snoozeActive = false;
    alarmStartMillis = millis();
    dfPlayer.volume(VOLUME_START);
    int nahodnaPisen = random(1, 6);  
    dfPlayer.loop(nahodnaPisen);
    redraw = true;   
  }

  // Automatické ukončení po minutě
  if (alarmActive && millis() - alarmStartMillis >= 60000) {
    alarmActive = false;
    snoozeCount++;
    if (snoozeCount < 3) {
      snoozeActive = true;
      nextAlarmMillis = millis() + (snoozeCount == 1 ? SNOOZE_1 : SNOOZE_2);
    } else {
      snoozeCount = 0;
      alarm1.enabled = false;
      saveAlarm();
      resetAfterAlarm();
    }
    dfPlayer.stop();
    dfPlayer.volume(0);
    stopAlarm = true;
  }

  // Plynulé zvyšování hlasitosti
  static unsigned long lastVolumeStep = 0;
  static uint8_t lastSetVolume = 255; // Paměť poslední odeslané hlasitosti

  if (alarmActive && millis() - lastVolumeStep > 100) {
    lastVolumeStep = millis();
    unsigned long elapsed = millis() - alarmStartMillis;
    uint8_t vol = VOLUME_START + (elapsed * (VOLUME_MAX - VOLUME_START)) / VOLUME_RAMP_TIME;
    if (vol > VOLUME_MAX) vol = VOLUME_MAX;
    
    if (vol != lastSetVolume) {
      dfPlayer.volume(vol);
      lastSetVolume = vol;
    }
  } else if (!alarmActive) {
    lastSetVolume = 255; // Reset paměti, když budík nehraje
  }

  // Pojistka jasu
  if (!wakeProcessActive && !snoozeActive && alarmBrightnessForced) {
    resetAfterAlarm();
  }

  // ===== TLAČÍTKA A PŘEKRESLOVÁNÍ =====
  bool nightNow = timeValid ? isNightNow(timeinfo) : false;
  
  // Tady se rozhodne, jestli kreslit 
  redraw = redraw || handleButtons(alarmActive, nightNow) || firstDraw || stopAlarm;

// ===== TIMEOUT PRO NASTAVOVÁNÍ BUDÍKU (5 sekund) =====
  if (uiState != SHOW_TIME) {
    if (millis() - lastSettingActivityMillis >= SETTING_TIMEOUT) {
      uiState = SHOW_TIME;
      userIsSettingAlarm = false; 
      saveAlarm();                
      lastAlarmMinute = -1;
      redraw = true;
    }
  }

  if (timeValid && timeinfo.tm_min != lastMinute) redraw = true;

  // POMALÉ BLIKÁNÍ V NASTAVENÍ (každých 500ms)
  if (uiState != SHOW_TIME) {
    static unsigned long lastBlink = 0;
    if (millis() - lastBlink > 500) {
      lastBlink = millis();
      redraw = true;
    }
  }

  // Timeout pro data z venkovního ESP32 (30 minut)
  // OPRAVENO: Přesunuto SEM před bránu vykreslování, aby vypršení okamžitě vyvolalo správný překresl displeje
  if (!isnan(tempExternal) && (millis() - lastExternalMillis > 30 * 60 * 1000UL)) {
    tempExternal = NAN;       // Změní teplotu na --,-
    batteryVoltage = NAN;     // Ikona baterie bude šedá
    redraw = true;            // Vynutíme okamžité překreslení
  }

// ===== Brána vykreslování: propustí kód jen při požadavku na změnu =====
  if (!redraw) return;
  redraw = false;      
  firstDraw = false;
  if (timeValid) lastMinute = timeinfo.tm_min;

  // ===== Vykreslení displeje =====
  if (uiState == SHOW_TIME) tft.fillScreen(TFT_BLACK);
  else tft.fillRect(80, 150, 160, 40, TFT_BLACK);

  // Datum
  tft.setTextSize(2);
  tft.setTextColor(TFT_WHITE);
  if (timeValid) {
    char dateStr[12];
    strftime(dateStr, sizeof(dateStr), "%d.%m.%Y", &timeinfo);
    tft.drawString(dateStr, 15, 15);
  } else tft.drawString("Cekam na cas...", 5, 5);

  // Ikona baterie
  drawBatteryIcon(270, 15, batteryVoltage);

  // Dny, svátky
  if (timeValid) {
    tft.drawString(dnyVTydnu[timeinfo.tm_wday], 235, 35);
    if (todayName.length()) tft.drawString(todayName, 15, 35);
  }

  // Teploty
  tft.setTextSize(2);

  // ==== vnitřní ====
  if (!isnan(tempInternal)) {
    char buf[20];
    tft.setTextColor(getIndoorTempColor(tempInternal), TFT_BLACK);
    snprintf(buf, sizeof(buf), "U postele: %.1f %cC", tempInternal, 247);		// nápis když zobrazuje měřenou hodnotu
    tft.drawString(buf, 15, 195);
  } else {
    char buf[20];
    tft.setTextColor(TFT_WHITE, TFT_BLACK);     					// barva když zobrazuje --:--
    snprintf(buf, sizeof(buf), "U postele: --,- %cC", 247); 				// nápis když zobrazuje --:--
    tft.drawString(buf, 15, 195);
  }

  // ==== venkovní ====
  if (!isnan(tempExternal)) {
    char buf[20];
    tft.setTextColor(getOutdoorTempColor(tempExternal), TFT_BLACK);
    snprintf(buf, sizeof(buf), "U krbu: %.1f %cC", tempExternal, 247);			// nápis když zobrazuje měřenou hodnotu
    tft.drawString(buf, 15, 215);
  } else {
    char buf[20];
    tft.setTextColor(TFT_WHITE, TFT_BLACK);    						 // barva když zobrazuje --:--
    snprintf(buf, sizeof(buf), "U krbu: --,- %cC", 247);  				// nápis když zobrazuje --:--
    tft.drawString(buf, 15, 215);       
  }

  // Čas
  if (timeValid) {
    char timeStr[6];
    strftime(timeStr, sizeof(timeStr), "%H:%M", &timeinfo);
    tft.setTextSize(7);
    tft.setTextColor(TFT_CYAN);
    tft.drawString(timeStr, 60, 90);
  } else {
    tft.setTextSize(4);
    tft.setTextColor(TFT_RED);
    tft.drawString("--:--", 60, 90);
  }

  // ==== Hláška při zvonění  ====
  if (alarmRunning) {
    tft.fillRect(0, 160, 320, 25, TFT_BLACK);  // smaže celý řádek

    tft.setTextSize(2);
    tft.setTextColor(TFT_RED, TFT_BLACK);       // barva písma
    tft.drawCentreString("Matyasku vstavej !!!", 160, 155, 2);   // text
  }

  // Budík s blikáním a stavem
  drawAlarmBlink(alarmRunning);

  // ==== watchdog
  esp_task_wdt_reset(); 
}
