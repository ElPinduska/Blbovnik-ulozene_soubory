#include <WiFi.h>
#include <HTTPClient.h>
#include <OneWire.h>
#include <DallasTemperature.h>
#include "esp_sleep.h"

// ===== WIFI =====
const char* ssid = "TP-LINK_30FF7A";
const char* password = "756123456";

// ===== CÍLOVÉ IP ADRESY =====
const char* targetIPs[] = {"192.168.0.232", "192.168.0.233"};
const int numTargets = sizeof(targetIPs) / sizeof(targetIPs[0]);

// ===== DS18B20 =====
#define ONE_WIRE_BUS 13           // měrení teploty nastaveno na pin 13 (signal data (prostredni pin))
OneWire oneWire(ONE_WIRE_BUS);
DallasTemperature sensors(&oneWire);

// ===== BATERIE =====
#define BATTERY_PIN 34
const float BAT_DIVIDER = 2.0;
const float ADC_MAX = 4095.0;
const float ADC_REF = 3.3;
const float BAT_CALIBRATION = 1.108;  // Kalibrační koeficient (3.47 skutečnost / 3.11 naměřeno) opravdová hodnota napětí : hodnota napetí co vypisuje ESP32

// ===== DEEP SLEEP =====
#define SLEEP_MINUTES 10

void setup() {
  Serial.begin(115200);
  delay(500);

  // ===== ADC =====
  analogReadResolution(12);
  analogSetPinAttenuation(BATTERY_PIN, ADC_11db);

  // ===== DS18B20 =====
  sensors.begin();

  // ===== WIFI =====
  WiFi.begin(ssid, password);
  Serial.print("WiFi...");
  unsigned long startAttempt = millis();
  while (WiFi.status() != WL_CONNECTED && millis() - startAttempt < 10000) {
    delay(500);
    Serial.print(".");
  }
  
  if (WiFi.status() != WL_CONNECTED) {
    Serial.println("\nWiFi chyba – jdu spat");
    goToSleep();
  }
  
  Serial.println("\nWiFi OK");
  delay(200);

  // ===== MĚŘENÍ =====
  sensors.requestTemperatures();
  float temp = sensors.getTempCByIndex(0);
  int raw = analogRead(BATTERY_PIN);
  float voltage = (raw / ADC_MAX) * ADC_REF * BAT_DIVIDER * BAT_CALIBRATION;
  
  Serial.printf("Temp: %.2f C | Bat: %.2f V\n", temp, voltage);

  // ===== ODESLÁNÍ NA VŠECHNY IP =====
  if (temp > -50 && temp < 85) {
    for (int i = 0; i < numTargets; i++) {
      sendData(targetIPs[i], temp, voltage);
    }
  }

  // ===== SLEEP =====
  goToSleep();
}

void loop() {
  // nepouziva se
}

// Pomocná funkce pro odeslání dat
void sendData(const char* ip, float t, float v) {
  HTTPClient http;
  
  // krátký timeout (4 sekundy stačí)
  http.setTimeout(4000); 
  String url = "http://" + String(ip) + "/update?temp=" + String(t, 1) + "&bat=" + String(v, 2);
  
  Serial.printf("Odesilam na %s... ", ip);
  
  //  Kontrola inicializace
  if (http.begin(url)) {
    int code = http.GET();
    if (code > 0) {
      Serial.printf("OK (kod: %d)\n", code);
    } else {
      Serial.printf("CHYBA (%s)\n", http.errorToString(code).c_str());
    }
    http.end();
  } else {
    Serial.println("Chyba inicializace HTTP");
  }
  
  //  Malá pauza před dalším pokusem v cyklu
  delay(100); 
}

void goToSleep() {
  Serial.println("Uspavam ESP32...");
  // zde se nastavuje čas spánku
  esp_sleep_enable_timer_wakeup((uint64_t)SLEEP_MINUTES * 60ULL * 1000000ULL);
  esp_deep_sleep_start();
}
